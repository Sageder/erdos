\begin{center}
\fbox{\parbox{0.92\textwidth}{\textbf{This note does NOT resolve Erd\H{o}s problem 289.}
The problem asks whether, for all sufficiently large $k$, one is a sum of the reciprocals of
$k$ pairwise disjoint blocks of consecutive integers, each of length at least $2$. What is
proved below is: the first block systems of this kind (none were previously exhibited here),
the exact minimum of their largest element, the truth of the statement for
$k=6,\dots,57$, and several structural theorems. The cofinite statement remains open, and
\S\ref{sec:open} states precisely the one analytic statement to which it has been reduced.}}
\end{center}

\section{Setting}

All letters denote integers and all reciprocal sums are exact elements of $\mathbb{Q}$.
A \emph{block} is an interval $[a,b]=\{a,a+1,\dots,b\}$ of positive integers with
$b>a\ge 1$; its length is $b-a+1\ge 2$ and its \emph{block sum} is
$H(a,b)=\sum_{n=a}^{b}1/n$. Blocks $[a_1,b_1]$ and $[a_2,b_2]$ are disjoint iff
$b_1<a_2$ or $b_2<a_1$; adjacent blocks are disjoint and count separately.

\begin{definition}
For $k\ge1$ let $P(k)$ be the statement: there exist $k$ pairwise disjoint blocks
$I_1,\dots,I_k$, each of length $\ge2$, with $\sum_{i=1}^{k}\sum_{n\in I_i}1/n=1$.
Erd\H{o}s problem 289 asks whether $P(k)$ holds for all sufficiently large $k$.
\end{definition}

Since a block of length $\ge2$ containing $1$ contributes at least $1+\tfrac12>1$ while all
other terms are positive, no block of a solution contains $1$; every element is $\ge2$.

\begin{definition}
A finite set $U\subseteq\mathbb{Z}_{\ge2}$ is \emph{legal} if it has no isolated point, i.e.
every $n\in U$ satisfies $n-1\in U$ or $n+1\in U$. Equivalently $U$ is a disjoint union of
blocks of length $\ge2$. Its \emph{maximal runs} are its maximal blocks of consecutive
integers; if these have lengths $L_1,\dots,L_r$ we write $r(U)=r$ (the run count) and
$M(U)=\sum_{i}\lfloor L_i/2\rfloor$ (the \emph{capacity}). A legal $U$ with
$\Sigma(U):=\sum_{n\in U}1/n=1$ is called a \emph{solution}.
\end{definition}

\begin{proposition}[Reformulation]\label{prop:reform}
$P(k)$ holds if and only if there is a solution $U$ with $r(U)\le k\le M(U)$.
\end{proposition}

\begin{proof}
$(\Leftarrow)$ A block of length $L$ can be partitioned into $t$ adjacent blocks each of
length $\ge2$ for every $1\le t\le\lfloor L/2\rfloor$: take $t-1$ parts of length $2$ and one
final part of length $L-2(t-1)\ge2$. Partitioning the $i$-th run of $U$ into $t_i$ parts,
with $1\le t_i\le\lfloor L_i/2\rfloor$ and $\sum_i t_i=k$ (possible exactly when
$r\le k\le M$), gives $k$ pairwise disjoint blocks of length $\ge2$ whose union is $U$, so
their reciprocals sum to $\Sigma(U)=1$.

$(\Rightarrow)$ Let $I_1,\dots,I_k$ be as in $P(k)$ and $U=\bigcup_i I_i$. Then $\Sigma(U)=1$
because the $I_i$ are disjoint, and $U$ is legal since each $I_i$ has length $\ge2$. Each
maximal run of $U$ is a disjoint union of some of the $I_i$, which are adjacent inside it;
a run of length $L$ can contain at most $\lfloor L/2\rfloor$ of them and at least one, so
$r(U)\le k\le M(U)$.
\end{proof}

Thus a single solution certifies $P(k)$ for every $k$ in the interval $[r(U),M(U)]$, and the
problem is equivalent to: \emph{do the intervals $[r(U),M(U)]$, over all solutions $U$, cover
a cofinite set of integers?}

\section{The $p$-adic conditions}

\begin{lemma}[Rule (P)]\label{lem:P}
Let $U$ be finite with $\Sigma(U)=q\in\mathbb{Q}$ and let $p$ be a prime. Then
$\nu_p\bigl(\sum_{n\in U,\,p\mid n}1/n\bigr)\ \ge\ \nu_p(q)$ whenever $\nu_p(q)\ge0$.
In particular, if $\Sigma(U)=1$ and $E=\max\{\nu_p(n):n\in U\}\ge1$, then, writing
$n=p^{E}m_n$ for the $n\in U$ attaining the maximum,
\[
\sum_{n\in U,\ \nu_p(n)=E} m_n^{-1}\ \equiv\ 0 \pmod p .
\]
Consequently at least two elements of $U$ attain $\nu_p=E$.
\end{lemma}

\begin{proof}
Write $\Sigma(U)=S_p+S'_p$ where $S_p$ sums $1/n$ over $n\in U$ with $p\mid n$ and $S'_p$
over the rest; every term of $S'_p$ lies in $\mathbb{Z}_p$, so $\nu_p(S'_p)\ge0$. If
$\nu_p(q)\ge0$ and $\nu_p(S_p)<0$ then $\nu_p(q)=\nu_p(S_p+S'_p)=\nu_p(S_p)<0$, a
contradiction. For the displayed congruence multiply $\Sigma(U)=1$ by $p^{E}$: in
$\mathbb{Z}_p$,
$p^{E}=\sum_{\nu_p(n)=E}m_n^{-1}+p\,(\text{something in }\mathbb{Z}_p)$, and $E\ge1$ forces
$\sum m_n^{-1}\equiv0\pmod p$. A single term $m^{-1}$ is a unit mod $p$, so the sum has at
least two terms.
\end{proof}

\begin{corollary}\label{cor:powersmooth}
If $\Sigma(U)=1$ and $N=\max U$, then every $n\in U$ satisfies $2p^{\nu_p(n)}\le N$ for every
prime $p\mid n$. In particular no element of $U$ is divisible by a prime $p>N/2$.
\end{corollary}

Taking $|U|$ with a unique maximal $\nu_2$ recovers the classical theorem of Kürschák: for
$b>a\ge1$, $H(a,b)\notin\mathbb{Z}$, because the unique multiple of the largest power $2^t$
occurring in $[a,b]$ gives $\nu_2(H(a,b))=-t\le-1$. Hence $P(1)$ is false.

\begin{theorem}[The top run is prime-free]\label{thm:toprun}
Let $U$ be a solution, $N=\max U$, and let $[c,N]$ be its top maximal run. Then $c>N/2$ and
$[c,N]$ contains no prime. Consequently the length of the top run is at most the largest gap
between consecutive primes not exceeding $N$.
\end{theorem}

\begin{proof}
If $c\le N/2$ then $[c,N]\supseteq(N/2,N]$, which contains a prime $p$ by Bertrand's
postulate. Then $p\in U$ and $\nu_p(p)=1$; any other multiple of $p$ is $\ge2p>N$ and so is
not in $U$, so $E=1$ is attained once, contradicting Lemma~\ref{lem:P}. Hence $c>N/2$, so
$[c,N]\subseteq(N/2,N]$ and the same argument forbids any prime in it.
\end{proof}

Theorem~\ref{thm:toprun} is the reason capacity cannot simply be manufactured by making the
last run long: the top run is confined to a prime gap.

\section{Certificates}

\begin{theorem}\label{thm:min}
There is no solution with $\max U\le84$. There are exactly four solutions with $\max U=85$,
namely (runs separated by $\mid$)
\[
\begin{array}{l}
5\,6\mid14\,15\mid17\,18\mid20\,21\,22\mid27\,28\mid33\,34\mid44\,45\mid54\,55\mid84\,85,\\
6\,7\,8\mid14\,15\mid17\,18\mid26\,27\mid34\,35\mid44\,45\mid54\,55\,56\mid65\,66\mid77\,78\mid84\,85,\\
7\,8\,9\,10\mid14\,15\mid17\,18\,19\mid34\,35\,36\mid44\,45\mid56\,57\mid76\,77\mid84\,85,\\
7\,8\mid11\,12\mid17\,18\mid21\,22\mid26\,27\,28\mid33\,34\mid44\,45\mid54\,55\,56\mid65\,66\mid77\,78\mid84\,85.
\end{array}
\]
In particular
\[
1=\tfrac15+\tfrac16+\tfrac1{14}+\tfrac1{15}+\tfrac1{17}+\tfrac1{18}+\tfrac1{20}+\tfrac1{21}
+\tfrac1{22}+\tfrac1{27}+\tfrac1{28}+\tfrac1{33}+\tfrac1{34}+\tfrac1{44}+\tfrac1{45}
+\tfrac1{54}+\tfrac1{55}+\tfrac1{84}+\tfrac1{85}.
\]
\end{theorem}

The statement is a finite computation, carried out exhaustively in exact integer arithmetic
by three independent programs (see \S\ref{sec:comp}); the four certificates were re-verified
from scratch with rational arithmetic.

\begin{theorem}\label{thm:k}
$P(k)$ holds for $k=6,7,\dots,57$.
\end{theorem}

\begin{proof}
By Proposition~\ref{prop:reform} it suffices to exhibit, for each such $k$, a solution $U$
with $r(U)\le k\le M(U)$; explicit block systems with exactly $k$ blocks are listed in
\texttt{experiments/CERTIFICATES.txt}, generated and re-verified by
\texttt{experiments/certificates.py} in exact rational arithmetic (count $=k$; every length
$\ge2$; pairwise disjoint; all elements $\ge2$; sum $=1$); the extreme cases were checked a
second and third time with \texttt{sympy.Rational} and with pure integer arithmetic
$\sum L/n = L$, $L=\mathrm{lcm}(U)$. For instance $k=9$ is certified by the first solution
of Theorem~\ref{thm:min}, and
\[
1=\Bigl(\tfrac14+\tfrac15+\tfrac16\Bigr)+\Bigl(\tfrac19+\tfrac1{10}\Bigr)
+\Bigl(\tfrac1{19}+\tfrac1{20}\Bigr)+\Bigl(\tfrac1{44}+\tfrac1{45}\Bigr)
+\Bigl(\tfrac1{132}+\tfrac1{133}\Bigr)+\Bigl(\tfrac1{209}+\tfrac1{210}\Bigr)
\]
certifies $k=6$, the smallest value for which $P(k)$ is currently known. \qedhere
\end{proof}

\begin{theorem}\label{thm:unit}
No block sum is a unit fraction: $H(a,b)\ne1/N$ for all $b>a\ge1$ and all $N\ge1$.
\end{theorem}

\begin{proof}[Proof sketch]
Let $k=b-a+1$ and let $2^{t}$ be the largest power of $2$ dividing an element of $[a,b]$;
as above $\nu_2(H(a,b))=-t$, so $2^{t}\mid N$ and $2^{t}\ge(k+1)/2$. Writing $n'$ for the
part of $n$ composed of primes $>k$, each such prime has a unique multiple in $[a,b]$, so
$R=\prod_{n=a}^{b}n'$ divides the reduced denominator; with
$\prod_{n=a}^{b}n=k!\binom{b}{k}$ and Kummer's bound $\nu_p\binom{b}{k}\le\log_p b$ one gets
$R\ge\binom{b}{k}b^{-\pi(k)}$. Combining $N<b/k$ with $2^{t}R\mid N$ yields a contradiction
whenever $k(k+1)\binom{b}{k}\ge2b^{1+\pi(k)}$, which holds for all $b\ge(k^2+k+2)/2$ once
$k\ge4$ except for finitely many pairs, checked exactly; $k=2$ and $k=3$ are direct.
\end{proof}

By contrast a \emph{system} of blocks can sum to a unit fraction; for instance
\[
\tfrac12=\tfrac16+\tfrac17+\tfrac1{20}+\tfrac1{21}+\tfrac1{44}+\tfrac1{45}
+\tfrac1{77}+\tfrac1{78}+\tfrac1{90}+\tfrac1{91},
\qquad
\tfrac13=\tfrac1{19}+\tfrac1{20}+\tfrac1{35}+\tfrac1{36}+\cdots+\tfrac1{104}+\tfrac1{105}.
\]

\section{What remains open}\label{sec:open}

For $T\ge2$ let $B(T)$ be the set of rationals expressible as $\Sigma(W)$ for a legal $W$ all
of whose elements are $\ge T$.

\begin{proposition}\label{prop:reduction}
Suppose $\tfrac12\in B(T)$ for arbitrarily large $T$ (equivalently: there are solutions
with $\min U\ge T$ for arbitrarily large $T$). Then $P(k)$ holds for all sufficiently large
$k$.
\end{proposition}

\begin{proof}[Proof sketch]
Fix the gadget $W_0\subseteq[6,91]$ with $\Sigma(W_0)=\tfrac12$ above. Given $T>91$ choose a
legal $W_T$ with elements $\ge T$ and $\Sigma(W_T)=\tfrac12$; then $U_T=W_0\cup W_T$ is a
solution. A legal set with elements $\asymp T$ and reciprocal sum $\tfrac12$ has $\asymp T$
elements, so $M(U_T)\to\infty$, while $r(U_T)\le M(U_T)$; provided the intervals
$[r(U_T),M(U_T)]$ chain (which holds as soon as $r$ stays a bounded fraction of $M$ and $M$
grows by bounded steps along a sequence of admissible $T$), they cover a cofinite set of $k$,
and Proposition~\ref{prop:reform} concludes.
\end{proof}

\begin{center}
\fbox{\parbox{0.92\textwidth}{\textbf{The open point.} Is $\tfrac12$ (or any fixed positive
rational) the reciprocal sum of a legal block system with all elements $\ge T$, for
arbitrarily large $T$?}}
\end{center}

This is the block-structured analogue of representing a rational by distinct unit fractions
with denominators confined to a short interval. Every mechanism examined in the underlying
investigation reduces to it, and several natural shortcuts are ruled out:

\begin{itemize}
\item Block sums are injective in the searched range: $H(a,b)=H(c,d)$ has no solution with
$(a,b)\ne(c,d)$ and elements $\le3000$. So a block cannot be replaced by a longer block of
equal sum (which would have raised capacity at fixed run count).
\item $\tfrac1a+\tfrac1{a+1}=\tfrac1c+\tfrac1{c+1}+\tfrac1d+\tfrac1{d+1}$ has no solution
with $a\le400$, $c\ge a+2$, $d\ge c+2$: an atom does not split into two atoms.
\item $B(T)$ is not shift-invariant: $\tfrac56\in B(2)$ but $\tfrac56\notin B(5)$ even
allowing elements $\le66$; $\tfrac7{12}\in B(3)\setminus B(4)$. On the other hand solutions
with $\min U\ge T$ have been found for every $T\le50$ tried, so the hypothesis of
Proposition~\ref{prop:reduction} is empirically well supported.
\item Parametric families of the shape ``main term plus small corrections'' fail for an
arithmetic reason. For $c\ge3$ the maps $n\mapsto\{cn,cn+1\}$ and $n\mapsto\{cn-1,cn\}$ give
pairwise disjoint atoms for every colouring, and the image realises exactly $k=|U|$ blocks;
but the coarse term $\sum_{n}(\tfrac1{cn}+\tfrac1{cn+1})$ has denominators divisible by $cn$
whereas the corrections $\tfrac{2}{c^2n^2-1}$ have denominators $(cn-1)(cn+1)$, coprime to
$cn$, so the corrections can never repair the coarse term. Blow-up families
$n\mapsto[qn-d+s_n,qn+d+s_n]$ fail similarly, on exactness.
\end{itemize}

\section{Computations}\label{sec:comp}

All searches use exact integer arithmetic over $L=\mathrm{lcm}$ of a pruned universe, with
weights $L/n$ and target $L$; no floating point occurs in any verification path. Two prunes
are used, both consequences of Lemma~\ref{lem:P}: a static one that iteratively deletes
integers that cannot lie in any solution, and a dynamic one requiring $Q[\mathrm{pos}]\mid R$
for the remaining value $R$, where $Q[\mathrm{pos}]$ pools the per-prime conditions still
available above $\mathrm{pos}$. The static prune is dramatic: the surviving universe of
$[2,300]$ has $174$ elements and lcm of only $68$ bits.

Exhaustive node counts for the plain engine: $79\,765$ at $N=40$, $1.4\cdot10^{6}$ at $50$,
$1.43\cdot10^{8}$ at $60$, $5.47\cdot10^{8}$ at $70$, $8.99\cdot10^{9}$ at $80$. An endgame
table over the top positions reduces $N=80$ to $1.9\cdot10^{6}$ nodes. Theorem~\ref{thm:min}
and Theorem~\ref{thm:k} rest on these searches together with independent re-verification of
every certificate.
